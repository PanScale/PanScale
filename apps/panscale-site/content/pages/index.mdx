## Quick start

```ts
import { createWebScaler } from "@panscale/web";

const scaler = createWebScaler(element, {
  contentWidth: 1600,
  contentHeight: 900,
  zooming: true,
  bouncing: true,
  callback: (values) => {
    content.style.transform =
      `matrix(${values.zoom},0,0,${values.zoom},${values.translateX},${values.translateY})`;
  },
});
```
